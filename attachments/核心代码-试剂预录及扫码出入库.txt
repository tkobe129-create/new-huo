## 三、数据库操作封装 (utils/db.js)

```javascript
import { getSupabase } from './supabase.js'
import { toDbFields, fromDbFields, toStockFields, fromStockFields } from './fields.js'

// 新增记录
export async function dbAdd(storeName, data) {
  const sb = getSupabase()
  let fields

  if (storeName === 'reagents') {
    fields = toDbFields(data)
  } else if (storeName === 'stockRecords') {
    fields = toStockFields(data)
  } else {
    fields = data
  }

  const { data: result, error } = await sb.from(storeName).insert(fields).select().single()
  if (error) throw error

  if (storeName === 'reagents') return fromDbFields(result)
  if (storeName === 'stockRecords') return fromStockFields(result)
  return result
}

// 更新记录
export async function dbPut(storeName, data) {
  const sb = getSupabase()
  let fields

  if (storeName === 'reagents') {
    fields = toDbFields(data)
  } else if (storeName === 'stockRecords') {
    fields = toStockFields(data)
  } else {
    fields = data
  }

  const { data: result, error } = await sb.from(storeName).update(fields).eq('id', data.id).select().single()
  if (error) throw error

  if (storeName === 'reagents') return fromDbFields(result)
  if (storeName === 'stockRecords') return fromStockFields(result)
  return result
}

// 获取单条记录
export async function dbGet(storeName, id) {
  const sb = getSupabase()
  const { data, error } = await sb.from(storeName).select('*').eq('id', id).single()
  if (error) throw error

  if (storeName === 'reagents') return fromDbFields(data)
  if (storeName === 'stockRecords') return fromStockFields(data)
  return data
}

// 获取所有记录
export async function dbGetAll(storeName) {
  const sb = getSupabase()
  const { data, error } = await sb.from(storeName).select('*').order('id', { ascending: true })
  if (error) throw error

  if (storeName === 'reagents') return data.map(fromDbFields)
  if (storeName === 'stockRecords') return data.map(fromStockFields)
  return data
}

// 删除记录
export async function dbDelete(storeName, id) {
  const sb = getSupabase()
  const { error } = await sb.from(storeName).delete().eq('id', id)
  if (error) throw error
}
```

```javascript
export function cleanFNC1(code) {
  return code
    .replace(/[\x1D]D/g, '')
    .replace(/[\x1D\]d]/gi, '')
    .replace(/\x1D/g, '')
}
```

### 4.4 解析条码（GS1 格式）

```javascript
/**
 * 格式: (AI)(data)(AI)(data)...
 * 示例输入: (01)(06954414719413)(11)(251206)(17)(270606)(10)(AB25100721H000502)
 */
export function parseBarcode(raw) {
  if (!raw || !raw.trim()) return []

  const parsedSegments = []
  const regex = /\((\d+)\)([^\(]*)/g
  let match

  while ((match = regex.exec(raw)) !== null) {
    const ai = match[1]
    const data = match[2]
    const start = match.index
    const end = start + match[0].length
    const info = GS1_AI_MAP[ai] || { desc: '未知(' + ai + ')', len: 0, color: 'seg-other' }
    parsedSegments.push({
      ai,
      desc: info.desc,
      color: info.color,
      start,
      end,
      value: data,
      fullMatch: match[0]
    })
  }

  if (parsedSegments.length === 0) {
    parsedSegments.push({
      ai: '',
      desc: '完整条码',
      color: 'seg-other',
      start: 0,
      end: raw.length,
      value: raw,
      fullMatch: raw
    })
  }

  return parsedSegments
}
```

### 4.5 根据切割规则提取字段

```javascript
/**
 * 从扫码结果中按预设规则提取批号、日期等
 * @param {string} scanResult - 扫码结果
 * @param {string} scanSegmentsStr - 切割规则 JSON
 * @returns {object} { lotNumber, productionDate, productionDateFormatted, expiryDate, expiryDateFormatted, gtin, serialNumber }
 */
export function extractFieldsByRules(scanResult, scanSegmentsStr) {
  if (!scanSegmentsStr) return {}
  try {
    const rules = JSON.parse(scanSegmentsStr)
    const fields = {}
    for (const rule of rules) {
      const value = scanResult.substring(rule.start, rule.start + rule.len)
      switch (rule.ai) {
        case '01':
          fields.gtin = value
          break
        case '10':
          fields.lotNumber = value
          break
        case '11':
          fields.productionDate = value
          if (value.length === 6) {
            fields.productionDateFormatted = '20' + value.substring(0, 2) + '-' + value.substring(2, 4) + '-' + value.substring(4, 6)
          }
          break
        case '17':
          fields.expiryDate = value
          if (value.length === 6) {
            fields.expiryDateFormatted = '20' + value.substring(0, 2) + '-' + value.substring(2, 4) + '-' + value.substring(4, 6)
          }
          break
        case '21':
          fields.serialNumber = value
          break
      }
    }
    return fields
  } catch (e) {
    return {}
  }
}
```

### 4.6 扫码匹配试剂（三级优先级）

```javascript
/**
 * 匹配优先级:
 *   1. scanRule 包含匹配（有GS1切割规则）
 *   2. code (GTIN) 包含匹配（有切割规则）
 *   3. 简单条码匹配（无切割规则，手动输入批号/效期）
 *
 * @param {string} scanResult - 扫码结果
 * @returns {object|null} { reagent, fields }
 */
export async function matchReagentByScan(scanResult) {
  const reagents = await dbGetAll('reagents')

  // 优先级 1: 通过 scanRule 匹配
  const withRule = reagents.filter(r => r.scanRule && r.scanRule.trim())
  for (const r of withRule) {
    if (scanResult.includes(r.scanRule)) {
      const fields = extractFieldsByRules(scanResult, r.scanSegments)
      return { reagent: r, fields }
    }
  }

  // 优先级 2: 通过 code (GTIN) 匹配
  const withCode = reagents.filter(r => r.code && r.code.trim())
  for (const r of withCode) {
    if (scanResult.includes(r.code)) {
      const fields = extractFieldsByRules(scanResult, r.scanSegments)
      return { reagent: r, fields }
    }
  }

  // 优先级 3: 简单条码匹配（无 GS1 结构）
  const simpleMatch = reagents.filter(r => r.code && r.code.trim() && !r.scanSegments)
  for (const r of simpleMatch) {
    if (scanResult === r.code || scanResult.includes(r.code)) {
      return { reagent: r, fields: {} }
    }
  }

  return null
}
```

```javascript
async function onScan(code) {
```

```javascript
function addToPending() {
  if (!matchedReagent.value) {
    uni.showToast({ title: '请先选择试剂', icon: 'none' })
    return
  }
  if (!form.value.quantity || Number(form.value.quantity) <= 0) {
    uni.showToast({ title: '请输入有效数量', icon: 'none' })
    return
  }

  // 出库时检查库存
  if (activeTab.value === 'out') {
    const currentStock = matchedReagent.value.stockQuantity || 0
    const alreadyPending = pendingList.value
      .filter(p => p.reagentId === matchedReagent.value.id)
      .reduce((sum, p) => sum + p.quantity, 0)
    if (Number(form.value.quantity) > currentStock - alreadyPending) {
      uni.showToast({ title: '出库数量超过可用库存', icon: 'none' })
      return
    }
  }

  pendingList.value.push({
    reagentId: matchedReagent.value.id,
    reagentName: matchedReagent.value.name,
    quantity: Number(form.value.quantity),
    lotNumber: form.value.lotNumber,
    expiryDate: form.value.expiryDate,
    productionDate: form.value.productionDate,
    note: form.value.note
  })

  uni.showToast({ title: '已添加', icon: 'success' })
  clearMatch()
}
```

### 5.3 批量提交入库/出库

```javascript
async function batchSubmit() {
  if (pendingList.value.length === 0) {
    uni.showToast({ title: '没有待提交的记录', icon: 'none' })
    return
  }

  try {
    uni.showLoading({ title: '提交中...' })

    for (const item of pendingList.value) {
      // 写入出入库记录
      await dbAdd('stockRecords', {
        reagentId: item.reagentId,
        type: activeTab.value,       // 'in' 或 'out'
        quantity: item.quantity,
        lotNumber: item.lotNumber,
        expiryDate: item.expiryDate,
        productionDate: item.productionDate,
        note: item.note,
        operator: '当前用户',
        department: '检验科',
        reason: ''
      })

      // 更新试剂库存
      const reagent = reagentList.value.find(r => r.id === item.reagentId)
      if (reagent) {
        if (activeTab.value === 'in') {
          reagent.stockQuantity = (reagent.stockQuantity || 0) + item.quantity
        } else {
          reagent.stockQuantity = Math.max(0, (reagent.stockQuantity || 0) - item.quantity)
        }
        await dbPut('reagents', reagent)
      }
    }

    uni.showToast({ title: `成功提交${pendingList.value.length}条记录`, icon: 'success' })
    pendingList.value = []
    await loadReagents()
    await loadHistory()
  } catch (err) {
    uni.showToast({ title: '提交失败', icon: 'none' })
  } finally {
    uni.hideLoading()
  }
}
```

### 6.2 自动解析条码生成切割规则

```javascript
function autoParseBarcode() {
  if (!barcodeRaw.value.trim()) return
  const segments = parseBarcode(barcodeRaw.value)
  parsedSegments.value = segments

  // 自动填充 scanRule（GTIN）
  const gtinSeg = segments.find(s => s.ai === '01')
  if (gtinSeg) {
    form.value.scanRule = gtinSeg.value
    form.value.code = form.value.code || gtinSeg.value
  }

  // 自动生成切割规则
  const rules = segments.map(s => {
    const dataStart = s.start + String(s.ai).length
    return { ai: s.ai, desc: s.desc, start: dataStart, len: s.end - dataStart }
  })
  form.value.scanSegments = rules
  scanSegmentsText.value = JSON.stringify(rules)
}
```

```javascript
async function saveReagent() {
  if (!form.value.code.trim()) {

    uni.showToast({ title: '保存失败: ' + (err.message || ''), icon: 'none' })
```

## 七、数据流向图

```
扫码枪/摄像头/粘贴
       │
       ▼
  cleanFNC1() 清理特殊字符
       │
       ▼
  matchReagentByScan() 三级匹配
       │
       ├── 优先级1: scanRule 包含匹配 → extractFieldsByRules() 自动提取
       ├── 优先级2: code 包含匹配     → extractFieldsByRules() 自动提取
       └── 优先级3: 简单条码匹配     → 返回空字段，手动输入
       │
       ▼
  pendingList 添加到待提交列表
       │
       ▼
  batchSubmit() 批量提交
       │
       ├── dbAdd('stockRecords', toStockFields(data))  写入出入库记录
       └── dbPut('reagents', reagent)                  更新库存数量
       │
       ▼
  Supabase PostgreSQL 数据库
       │
       ├── reagents 表（试剂信息）
       ├── stock_records 表（出入库记录）
       ├── settings 表（系统设置）
       └── users 表（用户信息）
```
